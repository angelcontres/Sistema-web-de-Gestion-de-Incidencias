\section{Perfil del Entorno \& Herramientas}
\textbf{Infraestructura de Pruebas:}
\begin{itemize}
    % Reemplazamos \texttt por \url para el dominio
    \item \textbf{Entorno de Despliegue:} Producción (Servidor VPS / Cloud hospedado \url{alertcity.dihm-muertos.site} vía HTTPS).
    \item \textbf{Base de Datos:} PostgreSQL en entorno de producción.
\end{itemize}

\textbf{Herramientas Utilizadas:}
\begin{itemize}
    % Usamos guiones sugeridos \- dentro de \texttt para la cabecera larga
    \item \textbf{Apache JMeter (v5.6.3):} Herramienta principal para simular la carga e inyectar peticiones HTTP estructuradas con cabeceras \texttt{Authori\-zation: Bearer} y \texttt{Accept: application/json}.
\end{itemize}

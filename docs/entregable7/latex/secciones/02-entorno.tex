\section{Perfil del Entorno \& Herramientas}

\textbf{Infraestructura de Pruebas:}
\begin{itemize}
    \item \textbf{Entorno de Despliegue:} Producción (Servidor VPS / Cloud hospedando \texttt{alertcity.dihm-muertos.site} vía HTTPS).
    \item \textbf{Base de Datos:} PostgreSQL en entorno de producción.
\end{itemize}

\textbf{Herramientas Utilizadas:}
\begin{itemize}
    \item \textbf{Apache JMeter (v5.6.3):} Herramienta principal para simular la carga e inyectar peticiones HTTP estructuradas con cabeceras \texttt{Authorization: Bearer} y \texttt{Accept: application/json}.
\end{itemize}
